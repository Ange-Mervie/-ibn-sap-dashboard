```ts
// ============================================================
// SLA simulé par priorité — le fichier Excel n'a AUCUNE colonne
// d'échéance : on calcule échéance = openDate + SLA_DAYS[priorité].
// Modifiez les seuils ci-dessous pour changer la définition du retard.
// ============================================================
import type { Issue } from "./issues";
import { isOpen, todayISO, daysBetween } from "./issues";

export const SLA_DAYS: Record<string, number> = {
  Critical: 3,
  High: 7,
  Medium: 14,
  Low: 30,
};

// « Échéance dans les 48 h » = alerte d'approche
export const WARNING_HOURS = 48;

// Statuts considérés comme bloqués / en attente
const BLOCKED_STATUSES = ["Blocked", "On Hold", "waiting for Customer"];

// Rang de priorité pour le tri des alertes
const PRIORITY_RANK: Record<string, number> = { Critical: 0, High: 1, Medium: 2, Low: 3 };

function addDays(iso: string, days: number): string {
  const d = new Date(iso + "T00:00:00");
  d.setDate(d.getDate() + days);
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

// Seuil SLA de la priorité (null = non évaluable)
export function slaDays(issue: Issue): number | null {
  return SLA_DAYS[issue.priority] ?? null;
}

// Échéance simulée : null si ticket sans date ou priorité inconnue
export function dueDate(issue: Issue): string | null {
  const days = slaDays(issue);
  if (!issue.openDate || days === null) return null;
  return addDays(issue.openDate, days);
}

// Heures restantes avant échéance (négatif = retard) ; null = non évaluable
export function hoursLeft(issue: Issue): number | null {
  const due = dueDate(issue);
  if (due === null) return null;
  return daysBetween(todayISO(), due) * 24;
}

// Jours de retard arrondis au supérieur (0 si pas de retard)
export function daysLate(issue: Issue): number {
  const h = hoursLeft(issue);
  if (h === null || h >= 0) return 0;
  return Math.ceil(-h / 24);
}

// Ouvert ET échéance dépassée
export function isLate(issue: Issue): boolean {
  const h = hoursLeft(issue);
  return isOpen(issue) && h !== null && h < 0;
}

// Ouvert ET échéance atteinte dans les WARNING_HOURS
export function isDueSoon(issue: Issue): boolean {
  const h = hoursLeft(issue);
  return isOpen(issue) && h !== null && h >= 0 && h <= WARNING_HOURS;
}

// --- agrégation des alertes du jour (sur TOUS les tickets) ---
export type AlertSummary = {
  late: Issue[]; // ouverts hors SLA, du plus en retard au moins en retard
  due: Issue[]; // ouverts à échéance ≤ 48 h, du plus proche au plus lointain
  blocked: Issue[]; // ouverts bloqués / en attente client, du plus ancien au plus récent
  criticalLate: number; // Critical en retard (pilote l'ouverture automatique du panneau)
  unique: number; // tickets distincts présents dans au moins un groupe
};

export function issueAlerts(issues: Issue[]): AlertSummary {
  const late = issues
    .filter(isLate)
    .sort((a, b) => daysLate(b) - daysLate(a) || (PRIORITY_RANK[a.priority] ?? 9) - (PRIORITY_RANK[b.priority] ?? 9));
  const due = issues
    .filter(isDueSoon)
    .sort((a, b) => (hoursLeft(a) ?? 0) - (hoursLeft(b) ?? 0));
  const blocked = issues
    .filter((i) => isOpen(i) && BLOCKED_STATUSES.includes(i.status))
    .sort((a, b) => (a.openDate || "").localeCompare(b.openDate || "")); // du plus ancien au plus récent
  const criticalLate = late.filter((i) => i.priority === "Critical").length;
  const unique = new Set([...late, ...due, ...blocked].map((i) => i.id)).size;
  return { late, due, blocked, criticalLate, unique };
}
```

```ts
// ============================================================
// Métriques de performance dans le temps (pures, sans Date du jour)
// Utilisées par la section « Performance et planning ».
// ============================================================
import type { Issue } from "./issues";
import { isOpen, resolutionDays } from "./issues";
import { slaDays } from "./sla";

function fmt(d: Date): string {
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

// Lundi de la semaine contenant `iso`
function mondayOf(iso: string): string {
  const d = new Date(iso + "T00:00:00");
  const weekday = (d.getDay() + 6) % 7; // 0 = lundi
  d.setDate(d.getDate() - weekday);
  return fmt(d);
}

// Clôturés par semaine (lundi), label dd/MM, tri chronologique
export function closedByWeek(issues: Issue[]): { name: string; value: number; from: string }[] {
  const weeks = new Map<string, number>();
  for (const i of issues) {
    if (!i.closeDate) continue;
    const from = mondayOf(i.closeDate);
    weeks.set(from, (weeks.get(from) || 0) + 1);
  }
  return [...weeks.entries()]
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([from, value]) => ({
      from,
      value,
      name: from.slice(8, 10) + "/" + from.slice(5, 7),
    }));
}

// Délai moyen de résolution par mois de clôture (mm/yy), tri chronologique
export function avgResolutionByMonth(issues: Issue[]): { name: string; value: number }[] {
  const months = new Map<string, { sum: number; n: number }>();
  for (const i of issues) {
    const d = resolutionDays(i);
    if (d === null || !i.closeDate) continue;
    const key = i.closeDate.slice(0, 7); // "YYYY-MM"
    const e = months.get(key) || { sum: 0, n: 0 };
    e.sum += d;
    e.n += 1;
    months.set(key, e);
  }
  return [...months.entries()]
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([key, e]) => ({
      name: key.slice(5, 7) + "/" + key.slice(2, 4), // mm/yy
      value: Math.round((e.sum / e.n) * 10) / 10,
    }));
}

// Taux de respect du SLA : clôturés avec les 2 dates et un SLA connu
export function slaRate(issues: Issue[]): { rate: number; onTime: number; base: number } {
  let onTime = 0;
  let base = 0;
  for (const i of issues) {
    if (isOpen(i)) continue;
    const d = resolutionDays(i);
    const s = slaDays(i);
    if (d === null || s === null) continue;
    base += 1;
    if (d <= s) onTime += 1;
  }
  return { rate: base ? Math.round((onTime / base) * 100) : 0, onTime, base };
}

// Taux de respect du SLA par groupe (responsable, module ou priorité)
export function slaRateBy(
  issues: Issue[],
  key: "raisedBy" | "module" | "priority"
): { name: string; rate: number; base: number }[] {
  const groups = new Map<string, Issue[]>();
  for (const i of issues) {
    const name = i[key] || "(non renseigné)";
    const g = groups.get(name) || [];
    g.push(i);
    groups.set(name, g);
  }
  return [...groups.entries()]
    .map(([name, list]) => ({ name, ...slaRate(list) }))
    .filter((g) => g.base > 0)
    .sort((a, b) => b.base - a.base || a.name.localeCompare(b.name));
}
```

```tsx
import { AlertTriangle, Clock, Lock, ChevronDown, ChevronUp } from "lucide-react";
import type { Issue } from "@/lib/issues";
import { ageDays } from "@/lib/issues";
import { issueAlerts, daysLate, dueDate, hoursLeft } from "@/lib/sla";
import { PRIORITY_COLORS } from "./Charts";

// Nombre de lignes du groupe « en retard » avant la bascule « Tout afficher »
const LATE_PREVIEW = 8;

function AlertLine({ issue, hint }: { issue: Issue; hint: string }) {
  const age = ageDays(issue);
  return (
    <li data-alert-row className="flex flex-wrap items-center gap-2 border-b border-zinc-100 py-1.5 text-xs last:border-0">
      <span className="font-mono text-zinc-500">n° {issue.id}</span>
      <span className="font-medium" style={{ color: PRIORITY_COLORS[issue.priority] || "#71717a" }}>
        {issue.priority}
      </span>
      <span className="min-w-0 flex-1 truncate text-zinc-700" title={issue.description}>
        {issue.description}
      </span>
      <span className="text-zinc-400">{issue.module || "sans module"}</span>
      <span className="text-zinc-500">{issue.raisedBy}</span>
      <span className="text-zinc-400">{age !== null ? `${age} j` : "—"}</span>
      <span className="rounded bg-red-50 px-1.5 py-0.5 font-semibold text-red-700">{hint}</span>
    </li>
  );
}

function Group({
  title,
  icon: Icon,
  color,
  group,
  issues,
}: {
  title: string;
  icon: React.ElementType;
  color: string;
  group: string;
  issues: Issue[];
}) {
  return (
    <div>
      <p className={`mb-1 flex items-center gap-1.5 text-xs font-semibold ${color}`}>
        <Icon size={13} /> {title} ({issues.length})
      </p>
      {issues.length === 0 ? (
        <p className="py-1 text-xs text-zinc-400">Aucun ticket dans cette catégorie.</p>
      ) : (
        <ul data-alert-group={group}>
          {issues.map((i) => (
            <AlertLine
              key={i.id}
              issue={i}
              hint={
                group === "blocked"
                  ? "bloqué"
                  : group === "due"
                    ? `J-${Math.max(0, Math.round((hoursLeft(i) ?? 0) / 24))}`
                    : `+${daysLate(i)} j`
              }
            />
          ))}
        </ul>
      )}
    </div>
  );
}

export default function Alerts({
  issues,
  open,
  onToggle,
}: {
  issues: Issue[];
  open: boolean;
  onToggle: () => void;
}) {
  const [showAllLate, setShowAllLate] = useState(false);
  const a = issueAlerts(issues);
  const lateShown = showAllLate ? a.late : a.late.slice(0, LATE_PREVIEW);

  return (
    <div className="mt-3">
      <button
        type="button"
        data-alerts-toggle
        onClick={onToggle}
        className="flex items-center gap-1.5 text-xs font-medium text-zinc-600 hover:text-zinc-900"
      >
        {open ? <ChevronUp size={13} /> : <ChevronDown size={13} />}
        {open ? "Masquer les détails des alertes" : "Afficher les détails des alertes"}
      </button>

      {open && (
        <div
          className="mt-2 grid grid-cols-1 gap-4 rounded-lg border border-zinc-200 bg-zinc-50 p-4 lg:grid-cols-3 print:break-inside-avoid"
        >
          <div>
            <Group
              title="En retard (SLA dépassé)"
              icon={AlertTriangle}
              color="text-red-700"
              group="late"
              issues={lateShown}
            />
            {a.late.length > LATE_PREVIEW && (
              <button
                type="button"
                onClick={() => setShowAllLate((v) => !v)}
                className="mt-1.5 text-xs font-medium text-blue-600 hover:underline"
              >
                {showAllLate
                  ? "Réduire"
                  : `… et ${a.late.length - LATE_PREVIEW} autres — Tout afficher (${a.late.length})`}
              </button>
            )}
            <p className="mt-2 text-[11px] text-zinc-400">
              {a.late.length}/{issues.filter((i) => i.status !== "Closed").length} ouverts sont hors SLA —
              le fichier ne contient aucun ticket récent.
            </p>
          </div>

          <Group
            title="Échéance dans les 48 h"
            icon={Clock}
            color="text-amber-700"
            group="due"
            issues={a.due}
          />

          <Group
            title="Bloqués / en attente client"
            icon={Lock}
            color="text-orange-700"
            group="blocked"
            issues={a.blocked.slice(0, 8)}
          />
        </div>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useMemo, useState } from "react";
import { CalendarCheck, Timer, ShieldCheck, AlertOctagon } from "lucide-react";
import type { Issue } from "@/lib/issues";
import { resolutionDays, PRIORITIES } from "@/lib/issues";
import { isLate, slaDays } from "@/lib/sla";
import { closedByWeek, avgResolutionByMonth, slaRate, slaRateBy } from "@/lib/metrics";
import { Panel, ChartBox, CountBars, HorizontalBars, TrendLine } from "./Charts";
import SectionHeader from "./SectionHeader";
import KpiCard from "./KpiCard";
import Gantt from "./Gantt"; // créé en Task 5 — si l'étape est exécutée avant, commenter le bloc Gantt (étape 5 bis)

const SEVEN_DAYS_MS = 7 * 86400000;

export default function Performance({ issues }: { issues: Issue[] }) {
  const [nature, setNature] = useState<"module" | "priority">("module");

  const stats = useMemo(() => {
    const limit = Date.now() - SEVEN_DAYS_MS;
    const closed7 = issues.filter(
      (i) => i.closeDate && new Date(i.closeDate + "T00:00:00").getTime() >= limit
    ).length;
    const delays = issues.map(resolutionDays).filter((x): x is number => x !== null);
    const avg = delays.length ? delays.reduce((a, b) => a + b, 0) / delays.length : null;
    const rate = slaRate(issues);
    const horsSla = issues.filter((i) => isLate(i)).length;
    return { closed7, avg, rate, horsSla };
  }, [issues]);

  const weeks = useMemo(() => closedByWeek(issues), [issues]);
  const months = useMemo(() => avgResolutionByMonth(issues), [issues]);
  const byOwner = useMemo(
    () => slaRateBy(issues, "raisedBy").map((g) => ({ name: `${g.name} (${g.base})`, value: g.rate })),
    [issues]
  );
  const byNature = useMemo(() => {
    const groups = slaRateBy(issues, nature);
    if (nature === "priority") {
      groups.sort((a, b) => PRIORITIES.indexOf(a.name as never) - PRIORITIES.indexOf(b.name as never));
    } else {
      groups.sort((a, b) => a.name.localeCompare(b.name));
    }
    return groups.map((g) => ({ name: `${g.name} (${g.base})`, value: g.rate }));
  }, [issues, nature]);

  const note =
    `Taux de respect = parmi les clôturés avec les 2 dates, part closeDate − openDate ≤ SLA ` +
    `(Critical ${slaDays({ priority: "Critical", openDate: "", id: "", description: "", status: "", raisedBy: "", module: "", closeDate: "", payment: "", comments: "" } as Issue)} j…) : ` +
    `valeur en %, (n) = clôturés avec dates du groupe. Responsable = « Raised By » : le fichier ne contient pas de responsable affecté.`;

  return (
    <div>
      <SectionHeader
        title="Performance et planning"
        subtitle="Évolution dans le temps, respect des délais par responsable et vue de planification"
        image="/banner/meeting.jpg"
      />

      {/* KPI de performance */}
      <div className="mb-6 grid grid-cols-2 gap-3 lg:grid-cols-4">
        <KpiCard
          label="Clôturés sur 7 jours"
          value={stats.closed7}
          sub="Close Date sur les 7 derniers jours"
          icon={CalendarCheck}
          color="text-blue-600"
        />
        <KpiCard
          label="Délai moyen"
          value={stats.avg !== null ? `${stats.avg.toFixed(1)} j` : "—"}
          sub={`${issues.length} tickets filtrés`}
          icon={Timer}
          color="text-zinc-900"
        />
        <KpiCard
          label="Taux de respect SLA"
          value={`${stats.rate.rate}%`}
          sub={`${stats.rate.onTime}/${stats.rate.base} clôturés dans les temps`}
          icon={ShieldCheck}
          color="text-green-600"
        />
        <KpiCard
          label="Tickets hors SLA"
          value={stats.horsSla}
          sub="Ouverts dont l'échéance est dépassée"
          icon={AlertOctagon}
          color="text-red-600"
        />
      </div>

      {/* Temps : hebdomadaire + tendance */}
      <div className="mb-6 grid grid-cols-1 gap-4 lg:grid-cols-2">
        <Panel title="Tickets clôturés par semaine (Close Date)">
          <ChartBox id="perf-week" data={weeks} emptyMsg="Aucune clôture pour ces filtres.">
            <CountBars data={weeks} color="#2563eb" order={weeks.map((w) => w.name)} />
          </ChartBox>
        </Panel>

        <Panel title="Délai moyen de résolution par mois de clôture">
          <ChartBox id="perf-trend" data={months} emptyMsg="Aucun délai calculable pour ces filtres.">
            <TrendLine data={months} />
          </ChartBox>
        </Panel>
      </div>

      {/* Respect du SLA : responsable + nature */}
      <div className="mb-6 grid grid-cols-1 gap-4 lg:grid-cols-2">
        <Panel title="Taux de respect SLA par responsable (Raised By)">
          <ChartBox id="perf-sla-owner" data={byOwner} emptyMsg="Aucun clôturé avec dates pour ces filtres.">
            <HorizontalBars data={byOwner} color="#16a34a" />
          </ChartBox>
          <p className="mt-2 text-xs text-zinc-400">{note}</p>
        </Panel>

        <Panel title="Taux de respect SLA par nature">
          <div className="mb-2 flex gap-1.5">
            <button
              type="button"
              data-nature="module"
              aria-pressed={nature === "module"}
              onClick={() => setNature("module")}
              className={`min-h-[36px] rounded-md border px-3 py-1.5 text-xs ${
                nature === "module"
                  ? "border-blue-600 bg-blue-600 text-white"
                  : "border-zinc-300 bg-white text-zinc-600 hover:bg-zinc-50"
              }`}
            >
              Module SAP
            </button>
            <button
              type="button"
              data-nature="priority"
              aria-pressed={nature === "priority"}
              onClick={() => setNature("priority")}
              className={`min-h-[36px] rounded-md border px-3 py-1.5 text-xs ${
                nature === "priority"
                  ? "border-blue-600 bg-blue-600 text-white"
                  : "border-zinc-300 bg-white text-zinc-600 hover:bg-zinc-50"
              }`}
            >
              Priorité
            </button>
          </div>
          <ChartBox id="perf-sla-nature" data={byNature} emptyMsg="Aucun clôturé avec dates pour ces filtres.">
            <HorizontalBars data={byNature} color="#2563eb" />
          </ChartBox>
          <p className="mt-2 text-xs text-zinc-400">{note}</p>
        </Panel>
      </div>

      {/* Planning Gantt (Task 5) */}
      <Panel title="Planning des incidents (ouverture → clôture)">
        <Gantt issues={issues} />
      </Panel>
    </div>
  );
}
```

```tsx
"use client";

import type { Issue } from "@/lib/issues";
import { isOpen } from "@/lib/issues";
import { isLate } from "@/lib/sla";
import { PRIORITY_COLORS } from "./Charts";

// ============================================================
// Planning type Gantt, pur (divs + CSS, aucune dépendance) :
// chaque ligne = un ticket, barre du début (openDate) à la fin
// (closeDate, ou aujourd'hui si encore ouvert).
// Barre rouge = encore ouvert ET hors SLA.
// ============================================================
const DAY_MS = 86400000;

function toMs(iso: string): number {
  return new Date(iso + "T00:00:00").getTime();
}

export default function Gantt({ issues }: { issues: Issue[] }) {
  if (!issues.length) {
    return <p className="py-8 text-center text-sm text-zinc-400">Aucun ticket à planifier pour ces filtres.</p>;
  }

  const todayMs = toMs(new Date().toISOString().slice(0, 10));
  const rows = [...issues].sort((a, b) => (a.openDate || "9999-99-99").localeCompare(b.openDate || "9999-99-99"));

  // Bornes de la timeline
  const starts = rows.map((r) => (r.openDate ? toMs(r.openDate) : null)).filter((x): x is number => x !== null);
  const ends = rows
    .map((r) => (r.closeDate ? toMs(r.closeDate) : r.openDate ? todayMs : null))
    .filter((x): x is number => x !== null);
  const min = starts.length ? Math.min(...starts) : todayMs;
  const max = Math.max(todayMs, ...(ends.length ? ends : [todayMs]));
  const span = Math.max(max - min, DAY_MS);
  const pct = (ms: number) => ((ms - min) / span) * 100;

  // Grille mensuelle (premier jour de chaque mois entre min et max)
  const grid: { label: string; left: number }[] = [];
  const cur = new Date(min);
  cur.setDate(1);
  while (cur.getTime() <= max) {
    grid.push({
      label: new Intl.DateTimeFormat("fr-FR", { month: "short", year: "2-digit" }).format(cur),
      left: pct(cur.getTime()),
    });
    cur.setMonth(cur.getMonth() + 1);
  }

  return (
    <div data-gantt>
      {/* axe des mois (décalé de la largeur de la colonne labels = w-28) */}
      <div className="relative ml-28 h-5 text-[10px] text-zinc-400">
        {grid.map((g, i) => (
          <span key={i} className="absolute -translate-x-1/2" style={{ left: `${g.left}%` }}>
            {g.label}
          </span>
        ))}
      </div>

      <div className="max-h-[420px] overflow-auto pr-1">
        <div className="relative min-w-[640px]">
          {/* grille mensuelle de fond */}
          <div className="pointer-events-none absolute inset-y-0 left-28 right-0 z-0">
            {grid.map((g, i) => (
              <div key={i} className="absolute inset-y-0 w-px bg-zinc-100" style={{ left: `${g.left}%` }} />
            ))}
          </div>

          {rows.map((r) => {
            const start = r.openDate ? toMs(r.openDate) : null;
            const end = r.closeDate ? toMs(r.closeDate) : r.openDate ? todayMs : null;
            const late = isOpen(r) && isLate(r);
            const left = start !== null ? pct(start) : 0;
            const width = start !== null && end !== null ? Math.max(pct(end) - pct(start), 0.4) : 0;
            const title = r.openDate
              ? `n° ${r.id} · ${r.priority} · du ${r.openDate} au ${r.closeDate || "aujourd'hui"}`
              : `n° ${r.id} · date d'ouverture absente`;
            return (
              <div key={r.id} data-gantt-row className="relative z-10 flex h-7 items-center border-b border-zinc-50">
                <span
                  className="w-28 shrink-0 truncate pr-2 font-mono text-[11px] text-zinc-500"
                  title={`n° ${r.id} — ${r.description}`}
                >
                  <span
                    className="mr-1 inline-block h-2 w-2 rounded-sm align-middle"
                    style={{ backgroundColor: PRIORITY_COLORS[r.priority] || "#94a3b8" }}
                  />
                  {r.id}
                </span>
                <div className="relative h-full flex-1">
                  {start !== null && end !== null ? (
                    <div
                      className={`gantt-bar absolute top-1/2 h-4 -translate-y-1/2 rounded-sm ${late ? "gantt-late" : ""}`}
                      style={{
                        left: `${left}%`,
                        width: `${width}%`,
                        backgroundColor: late ? "#dc2626" : PRIORITY_COLORS[r.priority] || "#94a3b8",
                      }}
                      title={title}
                    />
                  ) : (
                    <span className="absolute left-0 top-1/2 -translate-y-1/2 text-[10px] text-zinc-400">
                      Date absente
                    </span>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      </div>

      <p className="mt-2 flex flex-wrap gap-3 text-[11px] text-zinc-400">
        <span>Barre = openDate → closeDate (ou aujourd'hui si ouvert)</span>
        <span className="flex items-center gap-1">
          <span className="inline-block h-2 w-2 rounded-sm bg-red-600" /> ouvert hors SLA
        </span>
        {Object.entries(PRIORITY_COLORS).map(([name, color]) => (
          <span key={name} className="flex items-center gap-1">
            <span className="inline-block h-2 w-2 rounded-sm" style={{ backgroundColor: color }} /> {name}
          </span>
        ))}
      </p>
    </div>
  );
}
```

```tsx
  // ---- Filtres de période préréglés (d'après la date du jour) ----
  function isoDate(d: Date): string {
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, "0");
    const day = String(d.getDate()).padStart(2, "0");
    return `${y}-${m}-${day}`;
  }

  function presetRange(kind: "month" | "quarter"): { from: string; to: string } {
    const now = new Date();
    if (kind === "month") {
      return {
        from: isoDate(new Date(now.getFullYear(), now.getMonth(), 1)),
        to: isoDate(new Date(now.getFullYear(), now.getMonth() + 1, 0)),
      };
    }
    const quarterStart = Math.floor(now.getMonth() / 3) * 3;
    return {
      from: isoDate(new Date(now.getFullYear(), quarterStart, 1)),
      to: isoDate(new Date(now.getFullYear(), quarterStart + 3, 0)),
    };
  }

  function applyPreset(kind: "month" | "quarter" | "all") {
    if (kind === "all") setFilters((f) => ({ ...f, from: "", to: "" }));
    else setFilters((f) => ({ ...f, ...presetRange(kind) }));
  }

  // Pastille active = valeur des dates identique au préréglage
  const activePreset =
    filters.from || filters.to
      ? (["month", "quarter"] as const).find((k) => {
          const r = presetRange(k);
          return filters.from === r.from && filters.to === r.to;
        }) ?? null
      : null;
```
